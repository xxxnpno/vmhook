cmake_minimum_required(VERSION 3.28)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(vmhook_benchmarks LANGUAGES CXX)
endif()
option(VMHOOK_BENCHMARK_MODULE "Build the benchmark with import vmhook" OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
if(VMHOOK_BENCHMARK_MODULE)
    if(NOT TARGET vmhook::module)
        find_package(vmhook CONFIG REQUIRED COMPONENTS module)
    endif()
    set(_vmhook_benchmark_interface vmhook::module)
else()
    if(NOT TARGET vmhook::vmhook)
        find_package(vmhook CONFIG REQUIRED COMPONENTS header)
    endif()
    set(_vmhook_benchmark_interface vmhook::vmhook)
endif()
add_library(vmhook_benchmark SHARED runtime.cpp)
target_link_libraries(vmhook_benchmark PRIVATE ${_vmhook_benchmark_interface})
target_compile_options(vmhook_benchmark PRIVATE -Wall -Wextra -Werror)
target_link_options(vmhook_benchmark PRIVATE -static)
set_target_properties(vmhook_benchmark PROPERTIES PREFIX "")
if(VMHOOK_BENCHMARK_MODULE)
    target_compile_definitions(vmhook_benchmark PRIVATE VMHOOK_EXAMPLE_MODULE)
endif()
