cmake_minimum_required(VERSION 3.28)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(vmhook_examples LANGUAGES CXX)
endif()
option(VMHOOK_EXAMPLES_MODULE "Build examples with import vmhook" OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
if(VMHOOK_EXAMPLES_MODULE)
    if(NOT TARGET vmhook::module)
        find_package(vmhook CONFIG REQUIRED COMPONENTS module)
    endif()
    set(_vmhook_examples_interface vmhook::module)
else()
    if(NOT TARGET vmhook::vmhook)
        find_package(vmhook CONFIG REQUIRED COMPONENTS header)
    endif()
    set(_vmhook_examples_interface vmhook::vmhook)
endif()

foreach(_example IN ITEMS native_worker wrappers_collections method_hook field_class_observer)
    add_library(vmhook_example_${_example} SHARED ${_example}.cpp)
    target_link_libraries(vmhook_example_${_example} PRIVATE ${_vmhook_examples_interface})
    target_compile_options(vmhook_example_${_example} PRIVATE -Wall -Wextra -Werror)
    target_link_options(vmhook_example_${_example} PRIVATE -static)
    set_target_properties(vmhook_example_${_example} PROPERTIES PREFIX "")
    if(VMHOOK_EXAMPLES_MODULE)
        target_compile_definitions(vmhook_example_${_example} PRIVATE VMHOOK_EXAMPLE_MODULE)
    endif()
endforeach()
